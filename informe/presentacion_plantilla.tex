\begin{frame}{El punto de partida: la crisis de 2022}
\begin{columns}[T]
\begin{column}{0.55\textwidth}
\begin{itemize}
  \item En junio y julio de 2022, Cerro Prieto y La Boca bajaron a menos de 5\% de su capacidad.
  \item La ZMM tuvo entre 13,000 y 13,500 L/s disponibles frente a 16,500 L/s de consumo regular: un déficit de 3,000 a 3,500 L/s en el momento más crítico.
  \item Pronatura Noreste preguntó: \emph{¿dónde restaurar suelos en partes altas para favorecer la infiltración?}
  \item Nosotros agregamos la pregunta de fondo: \textbf{¿cuánta agua gana realmente el abasto?}
\end{itemize}
\end{column}
\begin{column}{0.4\textwidth}
\centering

\begin{frame}{Qué hicimos}
\small
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{enumerate}
  \item \textbf{Dónde:} índice de prioridad de siete criterios, con robustez ante 40 ponderaciones aleatorias.
  \item \textbf{Cuánta agua retienen las obras:} programa lineal con restricción de presupuesto sobre celdas de 1 km.
  \item \textbf{Dónde trazar las barreras:} \emph{minimum-cost maximum-flow} (MCMF) sobre la red de drenaje, con la capacidad de cada barrera.
  \item \textbf{Cuánta agua gana el abasto:} destino probabilístico del agua retenida con Monte Carlo.
\end{enumerate}
\end{column}
\begin{column}{0.48\textwidth}
\begin{enumerate}\setcounter{enumi}{4}
  \item \textbf{Presas:} erosión y azolve con la RUSLE.
  \item \textbf{Costo:} pesos por L/s frente a El Cuchillo II.
  \item \textbf{Vegetación:} especies nativas del tipo de vegetación de cada ladera (INEGI serie VII).
  \item \textbf{Validación:} calibración con el balance hídrico oficial de CONAGUA y proyecciones climáticas CMIP6.
\end{enumerate}
\end{column}
\end{columns}
\vspace{4pt}\clave{Datos abiertos (NASADEM, Sentinel-2, CHIRPS, MODIS, SMAP, HydroBASINS, INEGI, CONAGUA) y código reproducible; cada recomendación muestra sus reglas y datos.}

\begin{frame}{Las obras no desvían el agua: la retienen para que se infiltre}
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\begin{itemize}
  \item 3,000 ha retienen unos \textbf{«max_ls» L/s de agua bruta} (intervalo de 90\%: «mc_p5» a «mc_p95» L/s).
  \item El agua retenida sigue por el subsuelo dentro de la misma cuenca: una parte se evapotranspira, otra recarga el acuífero o llega a la presa como flujo base.
  \item Sin obras, parte de esa agua habría llegado a la presa de todos modos.
\end{itemize}
\clave{Ganancia neta por m³ retenido:\quad $G=(1-e)\,\phi-\tau$\\[2pt]
{\scriptsize $e$: evapotranspiración · $\phi$: fracción que llega a la fuente · $\tau$: lo que ya llegaba sin obras}}
\end{column}
\begin{column}{0.48\textwidth}
\centering\includegraphics[width=\linewidth]{figuras/g14_destino.png}
\nota{«fate_n» escenarios de Monte Carlo; la evapotranspiración se acota con MODIS.}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{En las presas, el beneficio es el suelo, no el volumen}

\begin{frame}{Por L/s neto, la restauración es más cara que el acueducto}

\begin{frame}{Tres algoritmos para calcular el agua retenida}
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\small
\begin{itemize}
  \item \textbf{\emph{Minimum-cost maximum-flow}} (\emph{network simplex}): exacto, admite costos por barrera.
  \item \textbf{\emph{Maximum flow} (Dinic)}: con nuestros costos es equivalente; mismo resultado, unas «orac_speed_dinic» veces más rápido.
  \item \textbf{\emph{Topological greedy fill}}: unas «orac_speed_fill» veces más rápido por evaluación, pero subestima la retención hasta «orac_fill_max», sobre todo en conjuntos optimizados.
  \item En la selección completa, Dinic elige las mismas barreras en «orac_sel_tiempo_dinic» del tiempo.
\end{itemize}
\end{column}
\begin{column}{0.47\textwidth}
\centering\includegraphics[width=\linewidth]{figuras/g16_oraculos_error.png}
\nota{«orac_nevals» evaluaciones en «orac_nsites» sitios y cuatro tormentas.}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{¿Cuánta agua se evapora detrás de las barreras?}
\begin{columns}[T]
\begin{column}{0.52\textwidth}
\small
\begin{itemize}
  \item \textbf{Satélites} (MODIS, PML, TerraClimate, FLDAS): «et_sc_min» a «et_sc_max» de la lluvia ordinaria se evapotranspira en la cuenca del río Santa Catarina. Es una cota superior.
  \item \textbf{Balance diario del suelo:} con el tratamiento completo cada metro recibe el escurrimiento de ~«vu_area» m²; «bk_e» del agua retenida se evapotranspira.
  \item Si se confirma, la ganancia neta con 30,000 ha pasa de «et_w0» a «et_w1» L/s (probabilidad positiva «et_p1»; con karst moderado, «et_pk»). El costo por L/s neto, «et_cost» M\$, sigue arriba de El Cuchillo II.
  \item Si parte del agua drenada se mueve ladera abajo dentro de la zona de raíces (flujo subsuperficial, raíces en el epikarst), $e\approx$ «if_e» y la ganancia baja a ~«if_w» L/s; con más de «if_g0» la ganancia esperada es negativa.
\end{itemize}
\end{column}
\begin{column}{0.45\textwidth}
\footnotesize
\textbf{Cómo medirla en campo}
\begin{itemize}
  \item Lisímetros de drenaje detrás de las barreras.
  \item Sensores de humedad a 30, 60 y 100 cm.
  \item Flujo de savia en árboles cercanos.
  \item Isótopos estables ($\delta^{18}$O, $\delta^2$H) y balance de cloruro.
  \item Covarianza de remolinos (\emph{eddy covariance}) para calibrar los satélites.
\end{itemize}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Qué tan sólidos son los resultados}
\footnotesize
\begin{itemize}
  \item \textbf{Calibración:} contra el escurrimiento natural oficial de Cerro Prieto (CONAGUA), el modelo sobrestima el escurrimiento; calibrado, el agua bruta y la neta bajan alrededor de «cal_drop», sin cambiar la probabilidad de ganancia positiva.
  \item \textbf{Acuífero:} la ganancia neta en la cuenca del río Santa Catarina es «aq_net_pct» de la recarga natural del acuífero Campo Buenos Aires: físicamente plausible.
  \item \textbf{Sequía:} con la lluvia de 2021 a 2022 se conserva «dr_pct» del agua bruta y «dr_overlap» de los sitios.
  \item \textbf{Clima 2041 a 2070:} la frecuencia de un año tan seco como 2021 a 2022 sería de «clima_f245» (SSP2-4.5) y «clima_f585» (SSP5-8.5), con mucha dispersión entre modelos.
  \item \textbf{Karst:} si parte del agua infiltrada sale por fallas a otras cuencas, la probabilidad de ganancia neta baja de «ks_base_p» a «ks_mod_p» con pérdida moderada («et_pk» con la evapotranspiración del balance del suelo).
  \item \textbf{Azolve de las barreras:} sin limpieza, el sedimento llena su almacenamiento en una o varias décadas y la ganancia neta bajaría de «vu_net0» a «vu_net1» L/s.
  \item \textbf{Limitación principal: a dónde va el agua infiltrada.} La evapotranspiración explica «lim_e_b» de la incertidumbre de la ganancia neta y el escurrimiento que ya llegaba «lim_t_b»; si el karst desvía agua, la ruta subterránea pasa a «lim_p_k» y es la más difícil de medir (trazadores). Los trazos de barreras son conceptuales (DEM de 30 m).
\end{itemize}
\end{frame}

\begin{frame}{Condiciones para escalar}
\begin{columns}[T]
\begin{column}{0.55\textwidth}
\footnotesize
\begin{enumerate}
  \item \textbf{Que el acuífero gane agua:} $G=(1-e)\phi-\tau>0$, es decir, evapotranspiración $e<1-\tau/\phi$ (umbral de 0.20 a 0.89 según $\phi$ y $\tau$; el balance del suelo estima $e$ de «bk_e» a «if_e_pct»).
  \item \textbf{Que la evidencia alcance:} intervalo de 90\% de $G$ medido en campo por encima de cero, con al menos 16 pares de microcuencas.
  \item \textbf{No como agua barata:} competir con El Cuchillo II exigiría $G\ge$ «esc_g25» a «esc_g50»; el máximo plausible es «esc_gmax».
  \item \textbf{Presas:} escalar sólo por el suelo y el azolve.
  \item \textbf{Que el agua llegue y las obras duren:} trazadores que confirmen la llegada al acuífero y limpieza periódica del sedimento.
  \item \textbf{Permisos:} CONANP y especies nativas de cada ladera.
\end{enumerate}
\end{column}

\begin{frame}{Recomendaciones}
\begin{enumerate}
  \item \textbf{Restaurar primero las laderas de la cuenca del río Santa Catarina}, donde el abasto sí gana agua, con autorización de la CONANP dentro del Parque Nacional Cumbres de Monterrey.
  \item \textbf{En las cuencas de las presas}, restaurar por el suelo, la vegetación y la vida útil de los embalses, sin esperar más agua.
  \item \textbf{Acompañar cada obra} sólo con especies nativas de su tipo de vegetación, de procedencia local y a densidad moderada.
  \item \textbf{Medir antes de escalar:} microcuencas pareadas con vertedores, humedad del suelo y piezómetros en los ocho sitios (unos 16 pares para detectar una diferencia de 50\%), más pares en Cerro Prieto.
  \item \textbf{Presentar la restauración como complemento}, no como sustituto de otras fuentes de abasto.
\end{enumerate}
\end{frame}
